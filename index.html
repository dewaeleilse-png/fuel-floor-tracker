const CACHE_NAME = "fuel-floor-tracker-v1"; const ASSETS = [ "./", "./index.html", "./manifest.json", "./service-worker.js", "./icon-192.png", "./icon-512.png" ]; self.addEventListener("install", (event) => { self.skipWaiting(); event.waitUntil( caches.open(CACHE_NAME).then((cache) => cache.addAll(ASSETS)) ); }); self.addEventListener("activate", (event) => { event.waitUntil( (async () => { const keys = await caches.keys(); await Promise.all(keys.map((k) => (k !== CACHE_NAME ? caches.delete(k) : null))); await self.clients.claim(); })() ); }); self.addEventListener("fetch", (event) => { event.respondWith( caches.match(event.request).then((cached) => cached || fetch(event.request)) ); });
